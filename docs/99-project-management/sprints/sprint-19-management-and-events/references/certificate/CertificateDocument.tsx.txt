/**
 * The attendance certificate (A4 landscape), design 07 "classic" of the SDC certificate template: a heavy double frame
 * with the identity mark in its four corners, the vertical colour logo, the community name in Arabic and English, a
 * centred statement, and a footer with the community leader's signature, the round emblem and the verification QR.
 * PDF colours cannot use CSS variables, so the values below are the written SDC identity colours (this file is exempt
 * from the token check). Arabic is set with IBM Plex Sans Arabic (shaped and right-to-left by the PDF engine); Latin
 * text uses Inter.
 */
const TEAL = '#2D7566';
const GREEN = '#58BF88';

/** Millimetres to PDF points. */
const mm = (v: number) => v * 2.8346;

/**
 * The PDF engine lays out an Arabic line that starts with a digit ("4 أكتوبر 2026") left to right, which puts the
 * day after the year. A leading RIGHT-TO-LEFT MARK fixes it. Only used for dates: it would move the « » quotes.
 */
const RLM = '\u200f';
const ARABIC = /[\u0600-\u06ff]/;
const rtl = (s: string) => (ARABIC.test(s) ? `${RLM}${s}` : s);

const SIGNER = { name: 'مهند الحربي', title: 'قائد المجتمع' };

const assets = (...p: string[]) => path.join(process.cwd(), 'public', 'assets', ...p);

let fontsReady = false;
function registerFonts() {
  if (fontsReady) return;
  Font.register({
    family: 'PlexArabic',
    fonts: [
      {
        src: assets('font', 'IBM_Plex_Sans_Arabic', 'IBMPlexSansArabic-Regular.ttf'),
        fontWeight: 400,
      },
      {
        src: assets('font', 'IBM_Plex_Sans_Arabic', 'IBMPlexSansArabic-Bold.ttf'),
        fontWeight: 700,
      },
    ],
  });
  Font.register({
    family: 'InterPdf',
    fonts: [
      { src: assets('font', 'Inter', 'static', 'Inter_18pt-Regular.ttf'), fontWeight: 400 },
      { src: assets('font', 'Inter', 'static', 'Inter_18pt-SemiBold.ttf'), fontWeight: 600 },
    ],
  });
  // Never hyphenate: Arabic words must not be split.
  Font.registerHyphenationCallback((word) => [word]);
  fontsReady = true;
}

/**
 * Name size: as large as fits on one line of the text column (about 600 pt), never below 22 pt; a very long name
 * wraps to a second line. Widths are estimated per character (Plex Arabic Bold: about 0.48 em Arabic, 0.6 em Latin).
 */
function nameSize(name: string): number {
  const ems = [...name].reduce((w, ch) => w + (ARABIC.test(ch) ? 0.48 : 0.6), 0);
  return Math.max(22, Math.min(34, Math.floor(600 / Math.max(ems, 1))));
}

/** The identity mark on a white square over each corner of the frame. */
function Corner({ at }: { at: { top?: number; bottom?: number; left?: number; right?: number } }) {
  return (
    <View style={[styles.corner, at]}>
      <Svg viewBox="0 0 70 61" width={mm(10)} height={mm(10 * (61 / 70))}>
        <Path d={MARK} fill={TEAL} />
      </Svg>
    </View>
  );
}

export type CertificateData = {
  id: string;
  recipientName: string;
  titleAr: string;
  titleEn: string | null;
  dateAr: string;
  dateEn: string;
  percent: number;
  sessionsAttended: number;
  sessionsExpected: number;
  verifyUrl: string;
  /** PNG data URL of the verification QR code. */
  qrDataUrl: string;
};

export function CertificateDocument({ c }: { c: CertificateData }) {
  registerFonts();
  const size = nameSize(c.recipientName);
  // Each fact is drawn as its own text (right to left): a run mixing Arabic, digits and "%" is reordered wrongly.
  const facts = [c.dateAr ? [rtl(c.dateAr)] : [], ['نسبة الحضور', `${c.percent}%`]].filter(
    (f) => f.length,
  );
  const factsEn = c.titleEn
    ? [c.titleEn, c.dateEn, `${c.percent}% attendance`].filter(Boolean).join('  ·  ')
    : null;
  const edge = mm(2);
  // The id is printed on its own line and the QR carries the full link, so the URL line shows only its base: the
  // full URL is too long for the space beside the emblem.
  const verifyBase = c.verifyUrl.endsWith(c.id) ? c.verifyUrl.slice(0, -c.id.length) : c.verifyUrl;
  return (
    <Document title={`Certificate ${c.id}`} author="Saudi Developer Community">
      <Page size="A4" orientation="landscape" style={styles.page}>
        <View style={styles.frame} />
        <View style={styles.frameInner} />
        <Corner at={{ top: edge, right: edge }} />
        <Corner at={{ top: edge, left: edge }} />
        <Corner at={{ bottom: edge, right: edge }} />
        <Corner at={{ bottom: edge, left: edge }} />

        <Image src={CERTIFICATE_LOGO} style={styles.logo} />
        <View style={[styles.side, { right: mm(26) }]}>
          <Text style={styles.sideAr}>المجتمع السعودي للمطورين</Text>
        </View>
        <View style={[styles.side, { left: mm(26) }]}>
          <Text style={styles.sideEn}>Saudi Developers Community</Text>
        </View>

        <View style={styles.head}>
          <Text style={styles.title}>شهادة حضور</Text>
          <View style={styles.titleEnRow}>
            <View style={styles.titleEnLine} />
            <Text style={styles.titleEn}>CERTIFICATE OF ATTENDANCE</Text>
            <View style={styles.titleEnLine} />
          </View>
        </View>

        <View style={styles.body}>
          <Text style={styles.statement}>يشهد المجتمع السعودي للمطورين بحضور</Text>
          <Text style={[styles.name, { fontSize: size }]}>{c.recipientName}</Text>
          <Svg width={mm(70)} height={2} style={styles.rule}>
            <Defs>
              <LinearGradient id="rule" x1="0" y1="0" x2="1" y2="0">
                <Stop offset="0" stopColor={GREEN} />
                <Stop offset="1" stopColor={TEAL} />
              </LinearGradient>
            </Defs>
            <Rect x="0" y="0" width={mm(70)} height={2} fill="url(#rule)" />
          </Svg>
          <Text style={styles.event}>«{c.titleAr}»</Text>
          <View style={styles.factsRow}>
            {facts.map((parts, i) => (
              <View key={parts.join()} style={styles.factItem}>
                {i > 0 ? <Text style={styles.dot}>·</Text> : null}
                {parts.map((p) => (
                  <Text key={p} style={/^[\d.]+%$/.test(p) ? styles.factsNum : styles.facts}>
                    {p}
                  </Text>
                ))}
              </View>
            ))}
          </View>
          {factsEn ? <Text style={styles.factsEn}>{factsEn}</Text> : null}
        </View>

        <View style={styles.signer}>
          <Image src={CERTIFICATE_SIGNATURE} style={styles.signature} />
          <View style={styles.signLine} />
          <Text style={styles.signName}>{SIGNER.name}</Text>
          <Text style={styles.signTitle}>{SIGNER.title}</Text>
        </View>

        <Image src={CERTIFICATE_EMBLEM} style={styles.emblem} />

        <View style={styles.verify}>
          <View style={styles.qrBox}>
            <Image src={c.qrDataUrl} style={styles.qr} />
          </View>
          <View style={styles.verifyText}>
            <Text style={styles.verifyLabel}>للتحقق من الشهادة · Verify this certificate</Text>
            <Text style={styles.verifyId}>{c.id}</Text>
            <Text style={styles.verifyUrl}>{verifyBase}</Text>
          </View>
        </View>
      </Page>
    </Document>
  );
}
